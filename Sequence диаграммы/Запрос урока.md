```plantuml
title Запрос урока (GET /courses/{course_id}/lessons/{lesson_id}/)

actor user as "Авторизованный пользователь"

participant apiGateway as "API Gateway"
participant contentService as "Сервис контента"
database contentService_db as "БД сервиса контента"
participant exercisesService as "Сервис заданий"
database exerciseService_db as "БД сервиса заданий"
participant videoService as "Сервис видеолекций"
database videoService_db as "БД сервиса видеолекций"

user -> apiGateway++: Запрос урока
apiGateway ->> contentService++: Запрос урока

contentService ->> contentService_db++: Запрос структуры урока, текстового и визуального контента
contentService <<-- contentService_db--: Ответ: структура урока, текстовый и визуальный контент

par
    opt При наличии практических заданий в уроке
        contentService -> exercisesService++: Запрос практических заданий
        exercisesService ->> exerciseService_db++: Запрос практических заданий
        exercisesService <<-- exerciseService_db--: Ответ: практические задания
        contentService <-- exercisesService--: Ответ: практические задания
    end
else
    opt При наличии видеолеккций в уроке
        contentService -> videoService++: Запрос видеолекций
        videoService ->> videoService_db++: Запрос видеолекций
        videoService <<-- videoService_db--: Ответ: видеолекции
        contentService <-- videoService--: Ответ: видеолекции
    end
end

apiGateway <<-- contentService--: Ответ: содержимое урока
user <-- apiGateway--: Ответ: содержимое урока
```
